## Provider Setup

The `GunboundSpacetimeProvider` sets up the SpacetimeDB connection and wraps the entire app:

```typescript
// src/features/game/spacetime/connection.tsx
import { useCallback, useMemo } from 'react';
import { SpacetimeDBProvider } from 'spacetimedb/react';
import { DbConnection } from './module_bindings';
import { readStoredToken } from '@/features/game/spacetime/token-storage';
import { SPACETIME_URI, SPACETIME_MODULE } from '@/features/game/spacetime/config';

export function GunboundSpacetimeProvider({ children }: { children: React.ReactNode }) {
  // Build connection once, reuse across renders
  const builder = useMemo(() => {
    const token = readStoredToken();
    
    return DbConnection.builder()
      .withUri(SPACETIME_URI)
      .withDatabaseName(SPACETIME_MODULE)
      .withToken(token || undefined)
      .onConnect(onConnect)
      .onConnectError(onConnectError);
  }, []);  // Empty deps — only run once

  return (
    <SpacetimeDBProvider connectionBuilder={builder}>
      {children}
    </SpacetimeDBProvider>
  );
}

function onConnect(conn: DbConnection) {
  // Subscribe to all public tables
  conn.subscriptionBuilder().subscribeToAllTables();
}

function onConnectError(err: Error) {
  console.error('SpacetimeDB connection failed:', err);
}
```

**Key points:**
- Use `useMemo` with empty deps to create the builder only once (prevents reconnects on render)
- Read stored token from localStorage on mount
- Pass token to builder so the connection uses the authenticated identity
- Subscribe to all tables in `onConnect` callback when you intentionally want one broad, non-cancelable public-table subscription

## Mounting the Provider

Wrap the root of your app:

```typescript
// src/app/page.tsx
import { GunboundSpacetimeProvider } from '@/features/game/spacetime';
import { HomeScreen } from '@/features/home/home-screen';

export default function Page() {
  return (
    <GunboundSpacetimeProvider>
      <HomeScreen />
    </GunboundSpacetimeProvider>
  );
}
```

All child components can now call SpacetimeDB hooks.

## useSpacetimeDB Hook

Access the connection from any component:

```typescript
import { useSpacetimeDB } from 'spacetimedb/react';

function MyComponent() {
  const connection = useSpacetimeDB();
  const conn = connection.getConnection();

  if (!conn) {
    return <div>Connecting...</div>;
  }

  // Use conn to call reducers
  async function handleClick() {
    await conn.reducers.myReducer({ param: 'value' });
  }

  return <button onClick={handleClick}>Do something</button>;
}
```

**Key checks:**
- `getConnection()` returns the connection or null if not yet established
- Always check for null before using the connection
- The connection is stable across renders (no need for useCallback deps)

## Identity and Authentication

When a user logs in, their SpacetimeDB `identity` is stored in the token. Access the current identity:

```typescript
function MyProfile() {
  const connection = useSpacetimeDB();
  const conn = connection.getConnection();

  // Get the current identity
  const identity = conn?.identity;

  if (!identity) {
    return <div>Not logged in</div>;
  }

  // Identity is an object with methods
  const identityHex = identity.toHexString();  // "abcd1234..."

  return <div>Logged in as {identityHex}</div>;
}
```

Use identity to find the current player:

```typescript
import { useTable } from 'spacetimedb/react';
import { tables } from '@/features/game/spacetime';

function PlayerStats() {
  const connection = useSpacetimeDB();
  const [players] = useTable(tables.player);
  const identity = connection.getConnection()?.identity;

  if (!identity) return <div>Not logged in</div>;

  const me = players.find(p =>
    p.identity.toHexString() === identity.toHexString()
  );

  if (!me) return <div>Loading...</div>;

  return (
    <div>
      <h1>{me.name}</h1>
      <p>Score: {me.score}</p>
      <p>XP: {me.xp}</p>
    </div>
  );
}
```

## useTable Hook

Subscribe to a table and receive real-time updates:

```typescript
import { useTable } from 'spacetimedb/react';
import { tables } from '@/features/game/spacetime';

function RoomList() {
  const [rooms, isReady] = useTable(tables.room);

  if (!isReady) return <div>Loading rooms...</div>;

  return (
    <ul>
      {rooms.map(room => (
        <li key={Number(room.id)}>
          {room.code} — {room.status}
        </li>
      ))}
    </ul>
  );
}
```

**Returns:**
- `[rows, isReady]` tuple
- `rows` is an array of rows from the table query
- `isReady` is true after the subscription has applied
- Re-renders automatically when data changes

`useTable` can receive a filtered query builder, not just a whole table:

```typescript
const roomQuery = tables.room.where((row) => row.code.eq('ABC12'));
const [rooms, isReady] = useTable(roomQuery);
```

## Composing Hooks

Combine multiple `useTable` calls to fetch related data:

```typescript
function RoomDetails({ roomId }: { roomId: bigint }) {
  const [rooms] = useTable(tables.room);
  const [members] = useTable(tables.roomMember);
  const [players] = useTable(tables.player);

  const room = rooms.find(r => r.id === roomId);
  const roomMembers = members.filter(m => m.roomId === roomId);
  
  // Resolve names
  const memberNames = roomMembers.map(m => {
    const player = players.find(p =>
      p.identity.toHexString() === m.identity.toHexString()
    );
    return player?.name ?? 'Unknown';
  });

  return (
    <div>
      <h2>{room?.code}</h2>
      <ul>
        {memberNames.map(name => <li key={name}>{name}</li>)}
      </ul>
    </div>
  );
}
```

## Custom Domain Hooks

Extract business logic into reusable hooks. Example: `useLobbyRooms`

```typescript
// src/features/lobby/spacetime/use-lobby-rooms.ts
import { useCallback, useMemo } from 'react';
import { useSpacetimeDB, useTable } from 'spacetimedb/react';
import { tables } from '@/features/game/spacetime';

export interface LobbyRoomView {
  id: bigint;
  code: string;
  status: string;
  hostName: string;
  memberCount: number;
  capacity: number;
}

export function useLobbyRooms() {
  const connection = useSpacetimeDB();
  const [dbRooms, roomsReady] = useTable(tables.room);
  const [players] = useTable(tables.player);

  // Filter non-ended rooms
  const activeRooms = useMemo(() => 
    dbRooms.filter(r => r.status !== 'ended'),
    [dbRooms]
  );

  // Transform to view model
  const rooms: LobbyRoomView[] = useMemo(() =>
    activeRooms.map(room => {
      const host = players.find(p =>
        p.identity.toHexString() === room.hostIdentity.toHexString()
      );
      return {
        id: room.id,
        code: room.code,
        status: room.status,
        hostName: host?.name ?? 'Unknown',
        memberCount: room.memberCount,
        capacity: room.capacity,
      };
    }),
    [activeRooms, players]
  );

  const createRoom = useCallback(async (code?: string) => {
    const conn = connection.getConnection();
    if (!conn) throw new Error('not connected');

    const roomCode = code || generateRoomCode();
    await conn.reducers.createRoom({ code: roomCode });
  }, [connection]);

  const joinRoomByCode = useCallback(async (code: string) => {
    const conn = connection.getConnection();
    if (!conn) throw new Error('not connected');

    await conn.reducers.joinRoomByCode({ code });
  }, [connection]);

  return {
    rooms,
    createRoom,
    joinRoomByCode,
    isReady: roomsReady,
  };
}

function generateRoomCode(): string {
  // Generate random 5-char code
  const alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // No 0, O, 1, I
  let code = '';
  for (let i = 0; i < 5; i++) {
    code += alphabet[Math.floor(Math.random() * alphabet.length)];
  }
  return code;
}
```

**Benefits:**
- Encapsulates data fetching and filtering
- Returns view models instead of raw DB rows
- Provides action methods (createRoom, joinRoomByCode)
- Simplifies component logic

## Error Handling

Wrap reducer calls in try-catch:

```typescript
async function handleJoinRoom(code: string) {
  try {
    await useLobbyRooms().joinRoomByCode(code);
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    showError(message);  // "room not found", "room is full", etc.
  }
}
```

## Loading States

Use the `isReady` flag to show initial loading UI. This is "subscription applied" readiness, not a generic network loading state:

```typescript
function RoomsList() {
  const { rooms, isReady } = useLobbyRooms();

  if (!isReady) {
    return <div>Loading rooms...</div>;
  }

  if (rooms.length === 0) {
    return <div>No rooms available. Create one!</div>;
  }

  return (
    <ul>
      {rooms.map(room => (
        <li key={Number(room.id)}>{room.code}</li>
      ))}
    </ul>
  );
}
```

## Performance Optimization

### Memoize Derived Data

```typescript
// ❌ SLOW — new object every render
const filtered = rooms.filter(r => r.status === 'waiting');

// ✅ FAST — memoized
const filtered = useMemo(() =>
  rooms.filter(r => r.status === 'waiting'),
  [rooms]
);
```

### Avoid Unnecessary Re-renders

Use `useCallback` for reducer calls to avoid stale closures:

```typescript
// ❌ BAD — new function every render, can cause stale state
const join = async (code) => {
  await conn.reducers.joinRoomByCode({ code });
};

// ✅ GOOD — stable reference
const join = useCallback(async (code) => {
  await conn.reducers.joinRoomByCode({ code });
}, [conn]);
```

### Extract Complex Logic

Move filtering and transformation to custom hooks:

```typescript
// ❌ BAD — multiple hooks and filtering in component
function Component() {
  const [rooms] = useTable(tables.room);
  const [players] = useTable(tables.player);
  const filtered = rooms.filter(...);
  const transformed = filtered.map(...);
  // Complex rendering logic
}

// ✅ GOOD — encapsulated in hook
function Component() {
  const { rooms } = useLobbyRooms();  // All logic is in the hook
  // Simple rendering
}
```

function ConnectionDebug() {

## Common Patterns

### Poll External Data

If you need to fetch data from an external API:

```typescript
function ExternalData() {
  const [data, setData] = useState(null);

  useEffect(() => {
    fetch('/api/some-endpoint')
      .then(r => r.json())
      .then(setData);
  }, []);

  return <div>{data}</div>;
}
```

SpacetimeDB doesn't directly support external HTTP calls, so use `fetch` or `axios` for external APIs.

### Real-time Notifications

Use the `onApplied` callback to detect when subscription is ready:

```typescript
const builder = useMemo(() => {
  return DbConnection.builder()
    .withUri(SPACETIME_URI)
    .withDatabaseName(SPACETIME_MODULE)
    .withToken(token || undefined)
    .onConnect((conn) => {
      conn.subscriptionBuilder()
        .onApplied(() => {
          console.log('Subscription ready — showing UI');
        })
        .onError((err) => {
          console.error('Subscription failed:', err);
        })
        .subscribe(["SELECT * FROM room WHERE status != 'ended'"]);
    });
}, []);
```

This fires once when the initial data is loaded.
